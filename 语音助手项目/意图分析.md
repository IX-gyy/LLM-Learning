# ai阿塔尼斯项目之意图分析模块
当前版本v2.0.2的意图分析非常呆板，流程大概是：用户输入→前端彩蛋关键词检测→后端关键词检测→GLM语义分析并决策走MCP或普通生成  
实践证明效果非常差，经常全部走普通路线，无法识别用户的需求。  
经过调研发现可以改进结构，具体改进结果可见我的仓库，这里主要介绍这个语义分类的模型。  
它实现语义分析的原理是先给出一系列例子，让模型将其向量化，用户再次输入语义类似的句子时，模型将其向量化后和已有的例子做比对，然后给出相似度，我们可以将高相似度的直接分为一类。  
这样的分类效果得到验证，比原先好了，重要的是例子一定要给足给全面，可以让ai帮你补全类似场景下用户可能说什么，这样可以提高成功率。  

## 探索模型
一开始Claude大人为我推荐了一个英文模型~~具体是哪个就不说了~~，但是因为它是英语语境下的，对中文语句的分类效果较差。  
后来推荐了bge-small-zh-v1.5，效果果然不错。下面是测试代码，可以参考一下：

# 模型本地路径

MODEL_PATH = os.path.join(BASE_DIR, "backend", "models", "bge-small-q8-zh-v1.5.gguf")

# ============================================================================
# 显式指令快速通道（和 keyword_filter.py 一致）
# ============================================================================
EXPLICIT_ROUTES = [
    (["提醒我", "别忘了", "记一下", "帮我记", "帮我加个提醒", "帮我加一个提醒"], "todo", {"sub_op": "add"}),
    (["待办列表", "有什么安排", "今天要做什么", "还有什么事", "我的任务"], "todo", {"sub_op": "list"}),
    (["天气", "会下雨", "会下雪", "温度多少", "气温多少", "台风"], "weather", {}),
    (["现在几点", "今天几号", "星期几", "现在什么时间", "当前时间"], "time_tool", {"sub_op": "current_time"}),
    (["电脑状态", "系统状态", "CPU占用", "内存使用", "磁盘空间", "电脑好卡", "电脑卡", "电脑慢", "电脑性能", "系统卡", "电脑怎么"], "system_status", {"sub_op": "all"}),
    (["帮我查", "搜索一下", "帮我搜", "帮我找"], "web_search", {}),
]

# ============================================================================
# 阈值
# ============================================================================
HIGH = 0.70
MEDIUM = 0.55
GAP = 0.08


def normalize(v):
    """L2 归一化"""
    norm = np.linalg.norm(v)
    if norm == 0:
        return v
    return v / norm


def cosine(a, b):
    return np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b))


def load_model():
    """使用 llama-cpp-python 加载 GGUF 嵌入模型"""
    print(f"模型路径: {MODEL_PATH}")

    if not os.path.isfile(MODEL_PATH):
        print(f"\n错误：GGUF 模型文件不存在！")
        print(f"请确认 {MODEL_PATH} 存在")
        sys.exit(1)

    print("正在使用 llama-cpp-python 加载 bge-small-q8-zh-v1.5.gguf 模型...")
    t0 = time.time()

    try:
        from llama_cpp import Llama
        # embedding=True 启用嵌入模式
        model = Llama(
            model_path=MODEL_PATH,
            n_ctx=512,
            n_threads=8,
            embedding=True,
            verbose=False
        )
        print(f"  加载成功, 耗时 {time.time()-t0:.1f}s")
        return model
    except ImportError:
        print("  llama-cpp-python 未安装")
        print("  pip install llama-cpp-python")
        sys.exit(1)
    except Exception as e:
        print(f"  加载失败: {e}")
        sys.exit(1)


def get_embedding(model, text):
    """使用 llama-cpp 获取文本嵌入向量"""
    output = model.create_embedding(text)
    embedding = output["data"][0]["embedding"]
    return np.array(embedding, dtype=np.float32)


def match_explicit(text):
    """显式指令匹配"""
    for keywords, tool_name, default_params in EXPLICIT_ROUTES:
        for kw in keywords:
            if kw in text:
                return (tool_name, default_params.copy())
    return None


def main():
    # 加载模型
    model = load_model()

    # 预计算各类别的平均嵌入
    print("正在预计算意图类别平均向量...")
    t0 = time.time()
    intent_embeddings = {}
    for intent, examples in INTENT_EXAMPLES.items():
        embeddings = []
        for text in examples:
            emb = get_embedding(model, text)
            emb = normalize(emb)
            embeddings.append(emb)
        intent_embeddings[intent] = np.mean(embeddings, axis=0)
    print(f"  完成, 耗时 {time.time()-t0:.1f}s\n")

    # 测试用例
    tests = [
        ("提醒我明天下午开会", "todo"),
        ("今天天气怎么样", "weather"),
        ("现在几点了", "time_tool"),
        ("帮我查量子计算", "web_search"),
        ("待办列表", "todo"),
        ("今天好累啊", "normal"),
        ("我明天要开会", "todo"),
        ("电脑好卡怎么回事", "system_status"),
        ("别忘了买酱油", "todo"),
        ("北京会下雨吗", "weather"),
        ("帮我倒计时五分钟", "time_tool"),
        ("今天心情真好", "normal"),
        ("电脑还剩多少内存", "system_status"),
        ("好困想去睡觉了", "normal"),
        ("深圳有没有暴雨预警", "weather"),
        ("还有多少天过年", "time_tool"),
        ("帮我找一下最近的新闻", "web_search"),
        ("今天晚上吃什么呢", "normal"),
    ]

    print(f"{'输入':22s} | {'期望':12s} | {'显式':30s} | {'语义 #1':20s} | {'#2':20s} | {'路由判断'}")
    print("-" * 140)

    correct_semantic = 0
    total_semantic = 0

    for text, expected in tests:
        # 显式路由
        explicit = match_explicit(text)
        explicit_str = f"{explicit[0]}" if explicit else "-"

        # 语义匹配
        t1 = time.time()
        query_emb = normalize(get_embedding(model, text))
        elapsed = (time.time() - t1) * 1000

        scores = {}
        for intent, avg_emb in intent_embeddings.items():
            scores[intent] = round(float(cosine(query_emb, avg_emb)), 3)
        scores = dict(sorted(scores.items(), key=lambda x: x[1], reverse=True))

        intents = list(scores.keys())
        best_intent = intents[0]
        best_score = scores[best_intent]
        second_intent = intents[1] if len(intents) > 1 else "-"
        second_score = scores[second_intent] if len(intents) > 1 else 0
        gap = round(best_score - second_score, 3)

        total_semantic += 1
        if best_intent == expected:
            correct_semantic += 1

        # 路由判断
        if best_score >= HIGH and gap >= GAP:
            route = f"→ {best_intent}"
        elif best_intent == "todo" and best_score >= MEDIUM:
            route = "→ TODO追问"
        elif gap < GAP and second_score >= MEDIUM:
            route = f"→ GLM({best_intent}|{second_intent})"
        else:
            route = "→ 正常对话"

        sn = f"{best_intent}={best_score:.3f}"
        s2 = f"{second_intent}={second_score:.3f}" if second_intent != "-" else "-"

        print(f"{text:22s} | {expected:12s} | {explicit_str:30s} | {sn:20s} | {s2:20s} | {route}")

    print("-" * 140)
    print(f"\n语义匹配准确率: {correct_semantic}/{total_semantic} = {correct_semantic/total_semantic*100:.1f}%")
    print(f"模型单次推理延迟: {elapsed:.1f}ms")


if __name__ == "__main__":
    main()
