```python
import torch
import numpy as np
from transformers import AutoModelForCausalLM, AutoTokenizer

# ========== 配置项 ==========
model_path = "./models/Qwen2.5-7B-Instruct"  # 替换为你的本地模型路径
device = "cuda:0"
# 统计的目标层：所有proj类Linear层，覆盖Attention和MLP的全部子层
target_layer_keywords = ["q_proj", "k_proj", "v_proj", "o_proj", 
                         "gate_proj", "up_proj", "down_proj"]
# 测试样本：覆盖不同领域，让激活分布更具代表性
sample_texts = [
    "请解释量子力学的基本原理",
    "计算：(12 + 34) * 5 - 78 / 3",
    "普通感冒期间有哪些注意事项",
    "介绍一下太阳系的八大行星",
    "用Python写一个快速排序算法",
    "分析一下全球气候变化的主要原因",
    "什么是大语言模型的量化技术",
    "请简述人体血液循环的过程",
    "翻译：Knowledge is power.",
    "列举5种常见的编程语言及其特点"
]

# ========== 加载模型 ==========
print("正在加载模型...")
model = AutoModelForCausalLM.from_pretrained(
    model_path,
    dtype=torch.float16,
    device_map=device,
    trust_remote_code=True
)
tokenizer = AutoTokenizer.from_pretrained(model_path, trust_remote_code=True)
model.eval()

# ========== 统计结果存储 ==========
activation_stats = {}

# ========== 统计钩子函数 ==========
def build_stat_hook(layer_full_name):
    def hook(module, input, output):
        # 取输出激活值，展平后统计
        act = output.detach().float().cpu().numpy().flatten()
        mean = float(np.mean(act))
        std = float(np.std(act))
        threshold_3sigma = mean + 3 * std
        # 离群值比例：大于mean+3σ的元素占比
        outlier_ratio = float(np.sum(act > threshold_3sigma) / len(act))
        
        activation_stats[layer_full_name] = {
            "mean": mean,
            "std": std,
            "threshold_3sigma": threshold_3sigma,
            "outlier_ratio": outlier_ratio
        }
    return hook

# ========== 注册钩子 ==========
hook_handles = []
for name, module in model.named_modules():
    # 只统计目标Linear层
    if isinstance(module, torch.nn.Linear):
        for kw in target_layer_keywords:
            if kw in name:
                handle = module.register_forward_hook(build_stat_hook(name))
                hook_handles.append(handle)
                break

print(f"已注册 {len(hook_handles)} 个统计钩子")

# ========== 前向传播采集激活 ==========
print("开始前向传播采集激活值...")
with torch.no_grad():
    for text in sample_texts:
        inputs = tokenizer(
            text, 
            return_tensors="pt", 
            truncation=True, 
            max_length=512
        ).to(device)
        model(**inputs)

# 移除钩子，释放资源
for handle in hook_handles:
    handle.remove()

# ========== 结果排序与输出 ==========
# 按离群值比例从高到低排序
sorted_layers = sorted(
    activation_stats.items(), 
    key=lambda x: x[1]["outlier_ratio"], 
    reverse=True
)

print("\n" + "="*80)
print("各层激活离群值比例排名（Top 20）")
print("="*80)
print(f"{'层名':<55} {'离群值比例':<12} {'均值':<10} {'3σ阈值':<10}")
print("-"*80)

for name, stat in sorted_layers[:20]:
    print(f"{name:<55} {stat['outlier_ratio']:<12.4f} {stat['mean']:<10.2f} {stat['threshold_3sigma']:<10.2f}")

# 按层类型汇总平均离群率
print("\n" + "="*80)
print("按层类型汇总平均离群值比例")
print("="*80)
type_stats = {kw: [] for kw in target_layer_keywords}
for name, stat in activation_stats.items():
    for kw in target_layer_keywords:
        if kw in name:
            type_stats[kw].append(stat["outlier_ratio"])
            break

for kw in target_layer_keywords:
    if type_stats[kw]:
        avg_ratio = np.mean(type_stats[kw])
        print(f"{kw:<15} 平均离群率: {avg_ratio:.4f}")

import json
with open("activation_stats.json", "w", encoding="utf-8") as f:
    json.dump(activation_stats, f, indent=2, ensure_ascii=False)
print("\n✅ 激活统计已保存到 activation_stats.json")
```

计算 4 个核心指标：
- mean：激活值的平均值
- std：标准差，反映数值的波动幅度
- threshold_3sigma：均值 + 3 倍标准差（统计学里的 3σ 阈值）
- outlier_ratio：大于 3σ 阈值的元素占总元素的比例 —— 这是我们判断 “离群值严不严重” 的核心指标。

> 为什么用 mean+3σ 当离群值标准？  
这是统计学里经典的 3σ 原则：  
如果数据符合正态分布，那么 99.73% 的数据都会落在 mean±3σ 区间内。也就是说，正常情况下大于 mean+3σ 的数值应该不到 0.3%。  
如果某层这个比例远高于 0.3%，就说明这层激活有大量极端离群值，分布非常不均匀。  

### 1.3策略设计
但是我们通过观察，发现和之前实验的结果有出入，这可能和使用的模型不同有关（原来使用的是base，现在改为更常见的`Qwen2.5-7B-Instruct`）SFT 训练会让模型的激活分布更平稳，MLP 输出层的离群值会被显著抑制。  
同时输入样本有限：我们只用了 10 条短文本做测试，序列长度短。down_proj 的离群值通常在长上下文、复杂推理输入下会更明显，短文本下不容易暴露。  
统计位置差异：不同研究统计的位置可能不同（有的统计激活函数前，有的统计后），我们统计的是 Linear 层的直接输出。  
不过这结果仍有意义，因为两个结果都指出**k_proj层是离群值较大的层**，那么我们的策略就是要跳过它。  
这里给出4个策略，它们将和原模型做对比：
1. full策略，全部进行int8量化  
2. 策略A：按激活均值排序，跳过Top 20%
3. 策略B：跳过所有的k_proj层
4. 策略C：跳过所有的k_proj和down_proj层（看看原实验是否有其过人之处）
下面给出具体代码：
```python
import torch
import json
from transformers import AutoModelForCausalLM

model_path = "./models/Qwen2.5-7B-Instruct"  # 替换为你的本地模型路径

# 加载模型（放CPU即可，仅用于获取层名，省显存）
model = AutoModelForCausalLM.from_pretrained(
    model_path, 
    dtype=torch.float16,
    device_map="cpu"
)

def get_layers_by_keyword(model, keywords):
    """根据关键词筛选所有匹配的Linear层，返回完整层名列表"""
    skip_layers = []
    for name, module in model.named_modules():
        if isinstance(module, torch.nn.Linear):
            for kw in keywords:
                if kw in name:
                    skip_layers.append(name)
                    break
    return skip_layers

# ========== 策略B：仅跳过k_proj（基于你的统计结果） ==========
skip_B = get_layers_by_keyword(model, ["k_proj"])
print(f"策略B跳层数量：{len(skip_B)}")

# ========== 策略C：k_proj + down_proj（完全对齐原报告） ==========
skip_C = get_layers_by_keyword(model, ["k_proj", "down_proj"])
print(f"策略C跳层数量：{len(skip_C)}")

# ========== 策略A：按激活均值排序，跳过Top 20% ==========
# 先把阶段1的激活统计结果保存为 activation_stats.json
# 如果你还没保存，可以把阶段1代码最后加上导出逻辑
with open("./results/activation_stats.json", "r", encoding="utf-8") as f:
    activation_stats = json.load(f)

# 按均值从大到小排序（原报告策略A为均值规则）
sorted_by_mean = sorted(
    activation_stats.items(), 
    key=lambda x: x[1]["mean"], 
    reverse=True
)
# 取前20%的层
n_total = len(sorted_by_mean)
n_skip = int(n_total * 0.2)
skip_A = [name for name, _ in sorted_by_mean[:n_skip]]
print(f"策略A跳层数量：{len(skip_A)}")

# 统一保存所有策略
strategies = {
    "full": [],                # 全量化，不跳任何层
    "strategy_A": skip_A,
    "strategy_B": skip_B,
    "strategy_C": skip_C
}

with open("skip_strategies.json", "w", encoding="utf-8") as f:
    json.dump(strategies, f, indent=2, ensure_ascii=False)

print("所有跳层策略已保存到 skip_strategies.json")
```

然后进行量化：
```python
import torch
import json
import os
from transformers import AutoModelForCausalLM, AutoTokenizer
from llmcompressor import oneshot
from llmcompressor.modifiers.quantization import QuantizationModifier
from datasets import Dataset  # 新增导入

# ========== 全局配置 ==========
MODEL_PATH = "./models/Qwen2.5-7B-Instruct"
OUTPUT_DIR = "./quant_models"
DEVICE = "cuda:0" 

# 加载校准集和跳层策略
with open("./datasets/general_calib.json", "r", encoding="utf-8") as f:
    CALIB_TEXTS = json.load(f)

with open("skip_strategies.json", "r", encoding="utf-8") as f:
    STRATEGIES = json.load(f)

# ========== 将校准文本转换为 Dataset ==========
calib_dataset = Dataset.from_dict({"text": CALIB_TEXTS})

# ========== 单模型量化函数 ==========
def quantize_single_model(strategy_name, skip_layers):
    print(f"\n{'='*60}")
    print(f"开始 LLM-Compressor 量化：{strategy_name}")
    print(f"跳层数量：{len(skip_layers)}")
    print(f"{'='*60}")

    # 1. 加载模型（使用 dtype 替代 torch_dtype）
    model = AutoModelForCausalLM.from_pretrained(
        MODEL_PATH,
        dtype=torch.float16,
        device_map="auto", 
        trust_remote_code=True
    )
    tokenizer = AutoTokenizer.from_pretrained(MODEL_PATH, trust_remote_code=True)
    model.eval()

    # 2. 配置量化算法 (Recipe)
    quant_modifier = QuantizationModifier(
        targets="Linear",          # 量化所有线性层
        scheme="W8A8",             # 权重量化为INT8，激活量化为INT8
        ignore=skip_layers         # 跳过指定层（层名列表）
    )

    # 3. 执行量化 (oneshot)
    oneshot(
        model=model,
        recipe=quant_modifier,
        dataset=calib_dataset,     # 传递 Dataset 对象
        tokenizer=tokenizer,
        text_column="text",        # 数据集中的文本列名
        max_seq_length=512,
        # 如果显存不足，可尝试启用 pipeline="sequential"
        # pipeline="sequential",
    )

    # 4. 保存量化模型
    save_path = os.path.join(OUTPUT_DIR, f"qwen2.5-7b-int8-{strategy_name}")
    model.save_pretrained(save_path, save_compressed=True)
    tokenizer.save_pretrained(save_path)
    print(f"量化完成，模型已保存到：{save_path}")

    # 释放显存
    del model
    torch.cuda.empty_cache()

# ========== 批量执行所有策略 ==========
if __name__ == "__main__":
    os.makedirs(OUTPUT_DIR, exist_ok=True)
    
    for name, skip_layers in STRATEGIES.items():
        quantize_single_model(name, skip_layers)
    
    print("\n✅ 所有策略量化全部完成！")
```

### 1.4结果解读
本次跳层量化实验验证了两个核心假设：
1. 全层INT8量化确实会带来精度损失，选择性跳过敏感层可以有效缓解精度下降；
2. 基于激活统计的结构化跳层策略（策略B/C），能在极少增加计算量的前提下，实现精度与量化覆盖率的最优折中。
其中**策略B（仅跳过k_proj层）整体表现最优**，在多数任务上追回了全量化的精度损失，部分任务甚至超过原生FP16基线（属于子集测试的正常波动），完全符合原报告“少量跳层即可大幅提升量化稳定性”的核心结论。

#### 三、相对变化对比（更直观看收益）
以「全层INT8量化」为基准，看三种跳层策略的精度提升：

| 策略 | 相对全量化的平均提升 | 优于全量化的任务数 | 核心特点 |
|------|----------------------|--------------------|----------|
| 策略A | -0.16个百分点 | 3/6 | 纯数据驱动跳层，效果不稳定，部分任务反而掉点 |
| 策略B | +0.63个百分点 | 4/6 | 仅跳k_proj一层，计算开销增加极少，综合性价比最高 |
| 策略C | +0.88个百分点 | 5/6 | 跳两层覆盖更全，常识/推理类任务提升明显，但MLP层放弃量化 |
